    32→function UtcClock(): React.ReactElement {
    33→  const [t, setT] = useState<number>(() => Date.now());
    34→  useEffect(() => {
    35→    const id = setInterval(() => setT(Date.now()), 1000);
    36→    return () => clearInterval(id);
    37→  }, []);
    38→  return <span className="font-mono text-[11px] text-zinc-300">{fClock(t)}</span>;
    39→}

    76→/** Live engine readout chip for the workspace breadcrumb rail (5 Hz).
    77→ *  Direction changes fire a one-shot price flash (green/red pulse).
    78→ *  Flash bookkeeping lives at module scope — same guardrail pattern as the
    79→ *  market engine (ms.inst): background state survives tab switches. */
    80→const TICKER_DIR = new Map<string, 'up' | 'dn'>();
    81→const TICKER_FLASH = new Map<string, number>();
    82→
    83→function TickerChip({ sym, label }: { sym: string; label?: string }) {
    84→  useRevision(); // 5 Hz re-render so direction flips are observed
    85→  const inst = ms.inst[sym];
    86→  if (!inst) return null;
    87→  const up = inst.changePct >= 0;
    88→  const d = up ? 'up' : 'dn';
    89→  if (TICKER_DIR.get(sym) !== d) {
    90→    TICKER_DIR.set(sym, d);
    91→    TICKER_FLASH.set(sym, (TICKER_FLASH.get(sym) ?? 0) + 1);
    92→  }
    93→  const flash = TICKER_FLASH.get(sym) ?? 0;
    94→  return (
    95→    <span className="hidden items-center gap-1.5 rounded border border-kborder2 bg-kpanel px-1.5 py-0.5 md:inline-flex" title={inst.def.name}>
    96→      <span className="font-mono text-[9px] tracking-wider text-zinc-500">{label ?? sym.replace('1!', '')}</span>
    97→      <span
    98→        key={flash}
    99→        className={`font-mono text-[10px] font-semibold tabular-nums ${up ? 'text-emerald-400' : 'text-rose-400'} ${flash > 0 ? (up ? 'flash-green' : 'flash-red') : ''}`}
   100→      >
   101→        {fPx(inst.last, inst.def.dec)}
   102→      </span>
   103→      <span className={`font-mono text-[9px] tabular-nums ${up ? 'text-emerald-400/80' : 'text-rose-400/80'}`}>
   104→        {fPct(inst.changePct)}
   105→      </span>
   106→    </span>
   107→  );
   108→}
